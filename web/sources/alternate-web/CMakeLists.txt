cmake_minimum_required(VERSION 3.20)
project(AlternateWebCores C CXX)
if(NOT EMSCRIPTEN OR NOT CORE_ROOT)
 message(FATAL_ERROR "Use Emscripten and provide CORE_ROOT")
endif()
file(TO_CMAKE_PATH "${CORE_ROOT}" CORE_ROOT)
add_compile_options("-ffile-prefix-map=${CORE_ROOT}=cores" "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=bridge")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VBA ${CORE_ROOT}/vba-next)
set(MELON ${CORE_ROOT}/melonds)
set(MELON_LIB ${MELON}/src/libretro)
set(COMMON ${VBA}/libretro-common)
file(READ ${VBA}/build/Makefile.common VBA_MAKE)
string(REGEX MATCHALL "\\$\\((CORE_DIR|LIBRETRO_COMM_DIR)\\)/[A-Za-z0-9_./-]+\\.c" VBA_FILES "${VBA_MAKE}")
string(REPLACE "$(CORE_DIR)" "${VBA}" VBA_FILES "${VBA_FILES}")
string(REPLACE "$(LIBRETRO_COMM_DIR)" "${COMMON}" VBA_FILES "${VBA_FILES}")
list(FILTER VBA_FILES EXCLUDE REGEX "/(thread|rthreads)\\.c$")
list(REMOVE_DUPLICATES VBA_FILES)
add_library(vba STATIC ${VBA_FILES})
target_include_directories(vba PRIVATE ${VBA} ${COMMON}/include)
target_compile_definitions(vba PRIVATE __LIBRETRO__ HAVE_HLE_BIOS INLINE=inline FRONTEND_SUPPORTS_RGB565 USE_CHEATS LOAD_FROM_MEMORY)

file(READ ${MELON}/Makefile.common MELON_MAKE)
string(FIND "${MELON_MAKE}" "ifeq ($(HAVE_OPENGL)" END_COMMON)
string(SUBSTRING "${MELON_MAKE}" 0 ${END_COMMON} MELON_MAKE)
string(REGEX MATCHALL "\\$\\((CORE_DIR|MELON_DIR)\\)/[A-Za-z0-9_./-]+\\.(cpp|c)" MELON_FILES "${MELON_MAKE}")
string(REPLACE "$(CORE_DIR)" "${MELON_LIB}" MELON_FILES "${MELON_FILES}")
string(REPLACE "$(MELON_DIR)" "${MELON}/src" MELON_FILES "${MELON_FILES}")
list(REMOVE_DUPLICATES MELON_FILES)
add_library(melon STATIC ${MELON_FILES})
target_include_directories(melon PRIVATE ${MELON}/src ${MELON_LIB} ${MELON_LIB}/libretro-common/include ${MELON}/src/teakra/include ${MELON}/src/teakra/src)
target_compile_definitions(melon PRIVATE __LIBRETRO__ MELONDS_VERSION="0.9.3")
target_compile_options(melon PRIVATE -fexceptions)
function(webcore target library source)
 add_executable(${target} ${source})
 target_include_directories(${target} PRIVATE ${COMMON}/include ${MELON}/src)
 target_link_libraries(${target} PRIVATE ${library})
 set_target_properties(${target} PROPERTIES SUFFIX ".js")
 target_link_options(${target} PRIVATE --no-entry -fexceptions -sMODULARIZE=1 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sENVIRONMENT=web,node "-sEXPORTED_FUNCTIONS=['_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAP16']")
endfunction()
webcore(vba-next vba gba-bridge.c)
target_compile_definitions(vba-next PRIVATE WEB_EXTENSION="gba")
target_link_options(vba-next PRIVATE -sEXPORT_NAME=createCore)
webcore(melonds melon "nds-bridge.c;battery.cpp;lid.cpp")
target_link_options(melonds PRIVATE -sEXPORT_NAME=createNDS)
