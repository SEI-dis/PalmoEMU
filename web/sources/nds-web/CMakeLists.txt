cmake_minimum_required(VERSION 3.20)
project(LocalNDS C CXX)
if(NOT EMSCRIPTEN OR NOT NDS_ROOT)
 message(FATAL_ERROR "Use the Emscripten toolchain and set NDS_ROOT to desmume2015/desmume")
endif()
file(TO_CMAKE_PATH "${NDS_ROOT}" NDS_ROOT)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(READ ${NDS_ROOT}/Makefile.common MAKE_SOURCE)
string(FIND "${MAKE_SOURCE}" "ifeq" FIRST_CONDITIONAL)
string(SUBSTRING "${MAKE_SOURCE}" 0 ${FIRST_CONDITIONAL} MAKE_SOURCE)
string(REGEX MATCHALL "\\$\\(CORE_DIR\\)/[A-Za-z0-9_./-]+\\.(cpp|c)" CORE_FILES "${MAKE_SOURCE}")
string(REPLACE "$(CORE_DIR)" "${NDS_ROOT}/src" CORE_FILES "${CORE_FILES}")
add_executable(core ${CORE_FILES} bridge.c battery.cpp lid.cpp)
target_include_directories(core PRIVATE ${NDS_ROOT}/src ${NDS_ROOT}/src/libretro ${NDS_ROOT}/src/libretro-common/include)
target_compile_options(core PRIVATE -O3 -fexceptions -Wno-c++11-narrowing -Wno-register -Wno-unused-command-line-argument)
target_compile_definitions(core PRIVATE NDEBUG)
set_target_properties(core PROPERTIES SUFFIX ".js")
target_link_options(core PRIVATE -O3 --no-entry -fexceptions -sMODULARIZE=1 -sEXPORT_NAME=createNDS -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sENVIRONMENT=web,node "-sEXPORTED_FUNCTIONS=['_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAP16']")
